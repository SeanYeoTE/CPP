OVERVIEW
--------
The Ford-Johnson algorithm is an optimal comparison-based sorting algorithm that minimizes
the number of comparisons needed. It works by pairing elements, recursively sorting pairs,
and then inserting remaining elements using the Jacobsthal sequence for optimal insertion order.

   ```cpp
   template <typename T>
   void PmergeMe<T>::fordJohnsonSort() {
       // Store original data
       _original = _data;
       
       // Handle edge cases
       if (_data.size() <= 1) return;
       
       // Start timing
       clock_t start = clock();
       
       // Execute Ford-Johnson steps
       createPairs();
       if (!_pairs.empty()) {
           recursiveSort(_pairs);
           std::vector<int> smaller = extractSmallerElements();
           insertRemainingElements(smaller);
       }
       
       // Handle odd element if exists
       if (_hasOddElement) {
           insertElement(_oddElement);
       }
       
       // Calculate timing
       clock_t end = clock();
       _time = static_cast<double>(end - start) / CLOCKS_PER_SEC * 1000000; // microseconds
   }
   ```

2.3 Pairing Phase (createPairs):
   ```cpp
   template <typename T>
   void PmergeMe<T>::createPairs() {
       _pairs.clear();
       _hasOddElement = false;
       
       // Process pairs
       for (size_t i = 0; i < _data.size(); i += 2) {
           if (i + 1 < _data.size()) {
               int first = _data[i];
               int second = _data[i + 1];
               // Store as (smaller, larger)
               if (first <= second) {
                   _pairs.push_back(std::make_pair(first, second));
               } else {
                   _pairs.push_back(std::make_pair(second, first));
               }
           } else {
               // Odd element
               _hasOddElement = true;
               _oddElement = _data[i];
           }
       }
   }
   ```

2.6 Insertion Phase (insertRemainingElements) - C++98 Compliant:
   ```cpp
   template <typename T>
   void PmergeMe<T>::insertRemainingElements(const std::vector<int>& remaining) {
       // Create main chain from larger elements
       _data.clear();
       for (std::vector<std::pair<int, int> >::const_iterator it = _pairs.begin(); 
            it != _pairs.end(); ++it) {
           _data.push_back(it->second);
       }
       
       // Generate Jacobsthal sequence for optimal insertion order
       std::vector<int> jacobsthal = generateJacobsthalSequence(static_cast<int>(remaining.size()));
       
       // Insert elements according to Jacobsthal sequence - C++98 style
       for (std::vector<int>::const_iterator it = jacobsthal.begin(); 
            it != jacobsthal.end(); ++it) {
           int index = *it;
           if (index >= 0 && index < static_cast<int>(remaining.size())) {
               int value = remaining[index];
               int pos = binarySearch(value, 0, static_cast<int>(_data.size()) - 1);
               _data.insert(_data.begin() + pos, value);
           }
       }
   }
   ```

   ```cpp
   template <typename T>
   int PmergeMe<T>::binarySearch(int value, int left, int right) {
       while (left <= right) {
           int mid = left + (right - left) / 2;
           if (static_cast<int>(_data[mid]) < value) {
               left = mid + 1;
           } else {
               right = mid - 1;
           }
       }
       return left;
   }
   ```

3.2 Jacobsthal Sequence (generateJacobsthalSequence) - C++98 Compliant:
   ```cpp
   template <typename T>
   std::vector<int> PmergeMe<T>::generateJacobsthalSequence(int n) {
       std::vector<int> jacobsthal;
       if (n <= 0) return jacobsthal;
       
       // Generate Jacobsthal numbers: J(0)=0, J(1)=1, J(n)=J(n-1)+2*J(n-2)
       std::vector<int> j;
       j.push_back(0);
       if (n > 0) j.push_back(1);
       
       for (int i = 2; j[i-1] < n; ++i) {
           int next = j[i-1] + 2 * j[i-2];
           if (next >= n) break;
           j.push_back(next);
       }
       
       // Create insertion order based on Jacobsthal sequence
       std::vector<bool> used(n, false);
       for (size_t i = 1; i < j.size(); ++i) {
           for (int idx = j[i] - 1; idx >= j[i-1]; --idx) {
               if (idx < n && !used[idx]) {
                   jacobsthal.push_back(idx);
                   used[idx] = true;
               }
           }
       }
       
       // Add remaining indices
       for (int i = 0; i < n; ++i) {
           if (!used[i]) {
               jacobsthal.push_back(i);
           }
       }
       
       return jacobsthal;
   }
   ```
